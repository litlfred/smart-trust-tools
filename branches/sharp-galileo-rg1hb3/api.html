API (WIT)

The API is defined first, in WIT, the WebAssembly Component Model interface language. Rust exports it from its WebAssembly component; Kotlin follows it through fixed mapping rules. Requirement tags such as [STT-COS-002] link into the graph explorer. For the same API as JavaScript callers see it, open the TypeScript reference, generated by jco and TypeDoc.

types.wit

source
package who:smart-trust@0.1.0;

/// Types shared by every interface.
///
/// WIT has no recursive types, so arbitrary CBOR never crosses the API
/// boundary as a tree: it crosses as encoded bytes (`cbor-bytes`). Typed
/// payloads (DVCMin, VHL) cross as records.
interface types {
    /// COSE algorithm. es256 = -7 (primary), ps256 = -37 (secondary).
    /// Both MUST be implemented. [STT-COS-002]
    enum cose-algorithm { es256, ps256 }

    /// 8-byte key identifier: first 8 bytes of SHA-256 over the DER DSC.
    /// [STT-KID-001]
    type kid = list<u8>;

    /// CWT NumericDate: integer seconds since the Unix epoch. [STT-CWT-002]
    type numeric-date = s64;

    /// Encoded CBOR data item.
    type cbor-bytes = list<u8>;

    /// Status of an hcert subclaim number. [STT-SEL-001, STT-SEL-003]
    enum subclaim-status {
        /// Allocated by WHO (>= 0).
        assigned,
        /// Listed by WHO as proposed (>= 0), not yet assigned.
        proposed,
        /// Negative: development range, free to use, not stable.
        development,
    }

    record subclaim {
        key: s32,
        status: subclaim-status,
        /// Canonical URL of the logical model, if any.
        logical-model: option<string>,
        title: string,
    }

    /// Bounds applied while decoding untrusted input.
    /// [STT-CBR-003, STT-ZIP-002]
    record decode-limits {
        max-input-chars: u32,
        max-inflated-bytes: u32,
        max-depth: u32,
        max-items: u32,
        max-string-bytes: u32,
    }

    /// QR error-correction level. Default q. [STT-QR-002]
    enum error-correction { l, m, q, h }

    enum severity { blocking, warning, information }

    record validation-issue {
        /// Element path, e.g. "vx.is".
        path: string,
        /// Stable machine code, e.g. "must-have-issuer-or-clinician-name".
        code: string,
        message: string,
        severity: severity,
    }

    /// Pipeline stages, used to say where verification failed.
    enum step {
        prefix,
        base45,
        inflate,
        cose,
        header,
        trust-lookup,
        signature,
        key-usage,
        cwt,
        time,
        hcert,
        payload,
    }

    /// Closed failure taxonomy, identical in Kotlin and Rust. [STT-VER-008]
    enum failure-reason {
        bad-prefix,
        unsupported-context-version,
        base45-invalid,
        inflate-failed,
        limit-exceeded,
        cose-malformed,
        header-invalid,
        unsupported-algorithm,
        kid-not-found,
        signature-invalid,
        key-usage-mismatch,
        expired,
        not-yet-valid,
        hcert-missing,
        multiple-subclaims,
        payload-invalid,
    }

    record diagnostic {
        step: step,
        severity: severity,
        message: string,
    }

    /// Sizes at each stage of issuance. [STT-QR-006]
    record encoding-report {
        cbor-bytes: u32,
        cose-bytes: u32,
        deflated-bytes: u32,
        hc1-chars: u32,
        qr-version: option<u8>,
        error-correction: option<error-correction>,
        warnings: list<string>,
    }
}

/// Steps 4, 6, 7, 8: CBOR, ZLIB, Base45 and the HC1 context identifier.
/// Pure functions: no crypto, no I/O, no clock.
interface codec {
    use types.{decode-limits, cbor-bytes};

    variant codec-error {
        /// Character outside the alphabet, dangling character or overflow.
        /// [STT-B45-002]
        base45-invalid(string),
        inflate-failed(string),
        /// A decode-limits bound was hit. [STT-CBR-003, STT-ZIP-002]
        limit-exceeded(string),
        cbor-malformed(string),
        /// No `HC[1-9A-Z]:` context identifier. [STT-HC1-001]
        bad-prefix,
        /// A well-formed context identifier other than HC1, e.g. "HC2".
        unsupported-context-version(string),
    }

    /// RFC 9285 Base45. [STT-B45-001]
    base45-encode: func(data: list<u8>) -> string;
    base45-decode: func(text: string) -> result<list<u8>, codec-error>;

    /// ZLIB (RFC 1950) wrapping DEFLATE (RFC 1951). [STT-ZIP-001]
    zlib-compress: func(data: list<u8>, level: u8) -> list<u8>;
    zlib-inflate: func(data: list<u8>, max-out: u32) -> result<list<u8>, codec-error>;

    /// Returns the context identifier version character (e.g. "1") and
    /// the remainder after the colon. [STT-HC1-001]
    detect-context: func(text: string) -> result<tuple<string, string>, codec-error>;

    /// Steps 6 to 8: COSE bytes to "HC1:..." string.
    to-hc1: func(cose: list<u8>) -> string;

    /// Steps 8 to 6 reversed: "HC1:..." string to COSE bytes.
    from-hc1: func(text: string, limits: decode-limits) -> result<list<u8>, codec-error>;

    /// Re-encode CBOR using RFC 8949 core deterministic encoding.
    /// [STT-CBR-001]
    cbor-deterministic: func(cbor: cbor-bytes, limits: decode-limits) -> result<cbor-bytes, codec-error>;

    /// RFC 8949 diagnostic notation, for display and debugging only.
    /// [STT-GEN-008]
    cbor-diagnostic: func(cbor: cbor-bytes, limits: decode-limits) -> result<string, codec-error>;
}

package who:smart-trust@0.1.0;

/// Step 5: COSE_Sign1, key identifiers and signature formats.
interface cose {
    use types.{cose-algorithm, kid, decode-limits};

    variant cose-error {
        malformed(string),
        /// Protected header failed type/length checks. [STT-VER-003]
        header-invalid(string),
        unsupported-algorithm(s32),
        /// ES256 signature not exactly 64 bytes, or bad DER. [STT-COS-005]
        bad-signature-format(string),
        signature-invalid,
        bad-certificate(string),
    }

    /// A parsed COSE_Sign1. `protected` is kept as received so the
    /// Sig_structure is rebuilt from the original bytes. [STT-COS-004]
    record sign1 {
        protected: list<u8>,
        payload: list<u8>,
        signature: list<u8>,
        alg: option<cose-algorithm>,
        /// The protected kid if present, else the unprotected kid.
        /// [STT-COS-003]
        kid: option<kid>,
        kid-in-protected: bool,
        /// Whether the input carried CBOR tag 18. [STT-COS-001]
        tagged: bool,
    }

    /// First 8 bytes of SHA-256 over the DER-encoded DSC. [STT-KID-001]
    kid-from-dsc: func(dsc-der: list<u8>) -> kid;

    enum base64-alphabet { standard, url-safe }

    /// [STT-KID-003]
    kid-to-base64: func(k: kid, alphabet: base64-alphabet) -> string;
    kid-from-base64: func(text: string) -> result<kid, cose-error>;

    parse-sign1: func(bytes: list<u8>, limits: decode-limits) -> result<sign1, cose-error>;

    /// CBOR of ["Signature1", protected, h'', payload].
    sig-structure: func(protected: list<u8>, payload: list<u8>) -> list<u8>;

    /// Verify a parsed COSE_Sign1 against one DSC. Low level: does not
    /// check EKU, time or trust. [STT-COS-002]
    verify-sign1: func(s: sign1, dsc-der: list<u8>) -> result<_, cose-error>;

    /// ASN.1 DER ECDSA-Sig-Value <-> fixed-width r||s. [STT-COS-005]
    ecdsa-der-to-raw: func(der: list<u8>) -> result<list<u8>, cose-error>;
    ecdsa-raw-to-der: func(raw: list<u8>) -> result<list<u8>, cose-error>;
}

package who:smart-trust@0.1.0;

/// Step 1: the subclaim registry.
interface subclaims {
    use types.{subclaim, subclaim-status};

    /// The IG defaults: 1, 3, 4, 5, -6, -7. [STT-SEL-001]
    defaults: func() -> list<subclaim>;

    /// >= 0 is WHO-reserved; negative is the development range.
    /// [STT-SEL-003]
    is-development-range: func(key: s32) -> bool;

    /// Mutable registry. Integrators register extra subclaims or override
    /// defaults, e.g. when DVC moves from -6 to an assigned number.
    /// [STT-SEL-002, STT-SEL-003]
    resource registry {
        /// Starts with `defaults()`.
        constructor();
        lookup: func(key: s32) -> option<subclaim>;
        register: func(s: subclaim);
        /// Which subclaim number carries the given payload kind.
        subclaim-for: func(kind: payload-kind) -> option<s32>;
        set-subclaim-for: func(kind: payload-kind, key: s32);
        all: func() -> list<subclaim>;
    }

    /// Payload kinds that have a typed codec. Others travel as raw CBOR.
    enum payload-kind { dvc-min, vhl }

    /// Inputs to the QR Code Generation 2.4 decision tree. [STT-SEL-005]
    record use-case-facts {
        ihr-entry-requirement: bool,
        must-verify-offline: bool,
        content-must-change-after-issue: bool,
        holder-controls-disclosure: bool,
        /// Estimated HC1 length, if known.
        estimated-hc1-chars: option<u32>,
    }

    enum advice { icvp, dvc, vhl }

    record payload-advice {
        advice: advice,
        reasons: list<string>,
    }

    advise: func(facts: use-case-facts) -> payload-advice;
}

/// Step 2: the DVCMin payload (trust-phw DVCMin / DVCMinPreQual).
interface dvc {
    use types.{validation-issue, cbor-bytes, decode-limits};

    /// DVCMinVaccineDetails. [STT-PAY-002]
    record dvc-min-vaccine-details {
        /// Vaccine product identifier. PreQual ID under DVCMinPreQual.
        vp: string,
        /// Date of vaccination (FHIR date).
        dv: string,
        /// Supervising clinician.
        cn: option<string>,
        /// Issuing organization identifier.
        is: option<string>,
        /// Batch or lot number.
        bo: string,
    }

    /// Everything in DVCMin except vx. Used to split a visit into one
    /// certificate per vaccine. [STT-PAY-003]
    record dvc-subject {
        n: string,
        dob: string,
        s: string,
        nt: string,
        id: option<string>,
        dt: option<string>,
        gn: option<string>,
        v: string,
    }

    /// DVCMin. vx is 1..1: one immunization event per certificate.
    /// [STT-PAY-001, STT-PAY-005]
    record dvc-min {
        n: string,
        dob: string,
        s: string,
        /// ISO 3166-1 alpha-3. [STT-PAY-007]
        nt: string,
        id: option<string>,
        dt: option<string>,
        gn: option<string>,
        vx: dvc-min-vaccine-details,
        v: string,
        /// Elements this version does not know, kept as CBOR map entries
        /// (key, encoded value). [STT-PAY-008]
        extensions: list<tuple<string, cbor-bytes>>,
    }

    enum dvc-profile {
        /// Unconstrained vp binding.
        dvc-min,
        /// vp MUST be a WHO PreQual product ID (required binding).
        dvc-min-prequal,
    }

    /// Cardinality, invariant (must-have-issuer-or-clinician-name), nt
    /// and vp checks. vp goes through the host terminology import.
    /// [STT-PAY-004, STT-PAY-006, STT-PAY-007]
    validate: func(p: dvc-min, profile: dvc-profile) -> list<validation-issue>;

    per-event: func(subject: dvc-subject, events: list<dvc-min-vaccine-details>) -> list<dvc-min>;

    to-cbor: func(p: dvc-min) -> cbor-bytes;
    from-cbor: func(cbor: cbor-bytes, limits: decode-limits) -> result<dvc-min, list<validation-issue>>;
}

/// The payload carried by an hcert entry, shared by issuer and verifier.
interface payloads {
    use types.{cbor-bytes};
    use dvc.{dvc-min};

    record raw-payload {
        subclaim: s32,
        cbor: cbor-bytes,
    }

    /// [STT-PAY-009]
    variant payload {
        dvc-min(dvc-min),
        /// Subclaim 5: URI of a Verifiable Health Link.
        vhl(string),
        /// Any subclaim without a typed codec (1, 3, 4, -7, custom).
        raw(raw-payload),
    }
}

/// Step 9: QR symbol. The core returns a module matrix; rendering is
/// optional. Camera scanning is out of scope. [STT-QR-003, STT-QR-005]
interface qr {
    use types.{error-correction};

    variant qr-error {
        /// Input has characters outside the QR Alphanumeric set.
        not-alphanumeric(string),
        too-long(u32),
    }

    record qr-matrix {
        /// Modules per side.
        size: u32,
        version: u8,
        error-correction: error-correction,
        /// Row-major, size*size entries, 1 = dark.
        modules: list<u8>,
    }

    /// Single Alphanumeric-mode segment, smallest version that fits
    /// unless pinned. [STT-QR-001, STT-QR-002]
    encode: func(hc1: string, ec: error-correction, pin-version: option<u8>) -> result<qr-matrix, qr-error>;

    record render-options {
        /// Whole device pixels per module, so modules stay square.
        module-px: u32,
        /// Quiet zone in modules, at least 4.
        quiet-zone: u32,
    }

    /// [STT-QR-004]
    to-svg: func(m: qr-matrix, opts: render-options) -> string;

    /// module-px giving a diagonal closest to `diagonal-mm` (35 to 60 mm
    /// recommended) at `dpi`. [STT-QR-004]
    module-px-for-diagonal: func(m: qr-matrix, diagonal-mm: f64, dpi: u32, quiet-zone: u32) -> u32;
}

package who:smart-trust@0.1.0;

/// Trust material. The store is filled by the host from whatever source it
/// fetched (GDHCN DID trustlist, national JWKS, pinned PEM), so the core
/// never does network I/O and stays synchronous. [STT-TRU-001, STT-GEN-012]
interface trust {
    use types.{kid, numeric-date};

    variant trust-error {
        parse(string),
        /// JWS proof on a DID document did not verify. [STT-TRU-003]
        proof-invalid(string),
        /// DSC not signed by a listed SCA, or AKI/SKI mismatch.
        /// [STT-TRU-006]
        chain-invalid(string),
    }

    /// Query for candidate DSCs. Optional fields narrow the match.
    record key-query {
        kid: kid,
        /// CWT iss (alpha-2). [STT-VER-010]
        iss: option<string>,
        domain: option<string>,
        participant: option<string>,
        usage: option<string>,
    }

    record trusted-dsc {
        der: list<u8>,
        kid: kid,
        country: option<string>,
        not-before: numeric-date,
        not-after: numeric-date,
        /// Extended Key Usage OIDs; empty means valid for any type.
        /// [STT-VER-004]
        extended-key-usages: list<string>,
        domain: option<string>,
        participant: option<string>,
    }

    resource trust-store {
        constructor();
        /// Each add-* returns the number of DSCs added.
        add-pem: func(pem: string) -> result<u32, trust-error>;
        add-der: func(der: list<u8>) -> result<u32, trust-error>;
        /// RFC 7517 JWK set with x5c. [STT-TRU-005]
        add-jwks: func(json: string) -> result<u32, trust-error>;
        /// GDHCN DID trustlist v2 document (embedded variant).
        /// [STT-TRU-002]
        add-did-document: func(json: string, verify-proof: bool) -> result<u32, trust-error>;
        /// SCA certificates for the optional chain check. [STT-TRU-006]
        add-sca-pem: func(pem: string) -> result<u32, trust-error>;
        /// Every DSC matching the kid (kids are not unique). [STT-KID-002]
        candidates: func(q: key-query) -> list<trusted-dsc>;
        size: func() -> u32;
        /// When the host last replaced the contents. [STT-TRU-004]
        loaded-at: func() -> option<numeric-date>;
        set-loaded-at: func(t: numeric-date);
        clear: func();
    }

    enum environment { dev, uat, prod }
    enum trustlist-variant { embedded, reference }

    /// URL of a GDHCN DID trustlist v2 document. None for a level means
    /// the "-" wildcard. [STT-TRU-002]
    did-trustlist-url: func(env: environment, which: trustlist-variant, domain: option<string>, participant: option<string>, usage: option<string>) -> string;
}

pipeline.wit

source
package who:smart-trust@0.1.0;

/// End-to-end issuance (steps 3 to 9).
///
/// Signing is two-phase so that any key holder can sign without the core
/// calling back into the host: prepare() returns the Sig_structure bytes,
/// the caller signs them with an HSM, KMS, Android Keystore or WebCrypto,
/// and finish() assembles the certificate. [STT-ISS-004, STT-COS-006]
interface issuer {
    use types.{cose-algorithm, kid, numeric-date, error-correction, encoding-report, validation-issue};
    use payloads.{payload};
    use qr.{qr-matrix};
    use subclaims.{registry};

    record validity {
        iat: numeric-date,
        /// Mandatory: exp is the only revocation mechanism. [STT-CWT-003]
        exp: numeric-date,
    }

    record issue-request {
        payload: payload,
        /// CWT iss, alpha-2. [STT-CWT-005]
        iss: option<string>,
        validity: validity,
        /// DSC whose key will sign; kid is derived from it. [STT-KID-001]
        dsc-der: list<u8>,
        alg: cose-algorithm,
    }

    record issuer-policy {
        /// Refuse negative subclaims unless true. [STT-SEL-003]
        allow-development-subclaims: bool,
        error-correction: error-correction,
        render-qr: bool,
        /// Verify the result against dsc-der before returning.
        /// [STT-ISS-002]
        self-verify: bool,
        /// Warn above this HC1 length, suggesting VHL. [STT-QR-006]
        hc1-warn-chars: u32,
    }

    variant issue-error {
        payload-invalid(list<validation-issue>),
        development-subclaim-not-allowed(s32),
        unknown-subclaim(s32),
        /// iat before or exp after the DSC validity. [STT-CWT-004]
        validity-outside-dsc,
        bad-certificate(string),
        bad-signature-format(string),
        self-verify-failed(string),
    }

    record issued-hcert {
        hc1: string,
        /// Raw COSE_Sign1 for 8-bit-safe transports. [STT-GEN-009]
        cose: list<u8>,
        qr: option<qr-matrix>,
        report: encoding-report,
    }

    resource pending-signature {
        /// Bytes to sign (CBOR Sig_structure). [STT-COS-004]
        to-be-signed: func() -> list<u8>;
        algorithm: func() -> cose-algorithm;
        kid: func() -> kid;
    }

    /// Phase 1: validate, build CWT, CBOR and protected header.
    prepare: func(req: issue-request, reg: borrow<registry>, policy: issuer-policy) -> result<pending-signature, issue-error>;

    /// Phase 2: attach the signature. ES256 accepts r||s or DER and always
    /// emits r||s. [STT-COS-005]
    finish: func(pending: pending-signature, signature: list<u8>) -> result<issued-hcert, issue-error>;

    /// One call, using the host `signer` import. [STT-ISS-001]
    issue: func(req: issue-request, reg: borrow<registry>, policy: issuer-policy) -> result<issued-hcert, issue-error>;
}

/// Verification (QR Code Generation section 11).
interface verifier {
    use types.{kid, numeric-date, decode-limits, cbor-bytes, step, failure-reason, diagnostic, cose-algorithm};
    use payloads.{payload};
    use trust.{trust-store, trusted-dsc};
    use subclaims.{registry};

    record verifier-policy {
        limits: decode-limits,
        /// Fail on more than one hcert entry (default true). [STT-SEL-004]
        strict-single-subclaim: bool,
        clock-skew-seconds: u32,
        /// Also check the DSC X.509 validity period. [STT-VER-006]
        check-dsc-validity: bool,
        /// Narrow DSC lookup by iss. [STT-VER-010]
        narrow-by-iss: bool,
        /// Check DSC EKU against the certificate type. [STT-VER-004]
        check-key-usage: bool,
    }

    record verified-hcert {
        iss: option<string>,
        iat: numeric-date,
        exp: numeric-date,
        subclaim: s32,
        payload: payload,
        /// The DSC that verified the signature.
        dsc: trusted-dsc,
        /// Other subclaims present (non-strict policy only).
        other-subclaims: list<s32>,
        /// Unknown CWT claims, as (encoded key, encoded value).
        /// [STT-CWT-006]
        other-claims: list<tuple<cbor-bytes, cbor-bytes>>,
        diagnostics: list<diagnostic>,
    }

    record verification-failure {
        step: step,
        reason: failure-reason,
        detail: string,
        diagnostics: list<diagnostic>,
    }

    /// Payload content is only reachable through `verified`.
    /// [STT-VER-002, STT-VER-008]
    variant verification-result {
        verified(verified-hcert),
        failed(verification-failure),
    }

    /// `now` is passed in: no ambient clock. [STT-VER-005]
    verify: func(hc1: string, store: borrow<trust-store>, reg: borrow<registry>, now: numeric-date, policy: verifier-policy) -> verification-result;

    /// Same, for raw COSE_Sign1 bytes. [STT-GEN-009]
    verify-cose: func(cose: list<u8>, store: borrow<trust-store>, reg: borrow<registry>, now: numeric-date, policy: verifier-policy) -> verification-result;

    /// Decoded but NOT verified. Diagnostics only; never use for
    /// decisions. [STT-VER-002]
    record unverified-view {
        alg: option<cose-algorithm>,
        kid: option<kid>,
        claims: cbor-bytes,
    }

    inspect-unverified: func(hc1: string, limits: decode-limits) -> result<unverified-view, verification-failure>;
}

/// Implemented by the host (imported by the component).

/// Key holder for the one-call `issuer.issue`. Hosts with asynchronous
/// signers (WebCrypto, remote KMS) use issuer.prepare/finish instead.
/// [STT-COS-006]
interface signer {
    use types.{cose-algorithm};

    algorithm: func() -> cose-algorithm;
    /// ES256: r||s or DER; PS256: raw signature.
    sign: func(to-be-signed: list<u8>) -> result<list<u8>, string>;
}

/// Value-set membership, e.g. PreQual product IDs for DVCMinPreQual.
/// [STT-PAY-006]
interface terminology {
    enum membership { member, not-member, unknown }

    is-member: func(value-set: string, code: string) -> membership;
}
