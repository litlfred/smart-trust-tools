Kotlin API

Declarations only, with no behaviour, implemented by hcert-core. tools/wit2kotlin/wit2kotlin.py generates them from the contract, so the Kotlin and Rust libraries cannot drift apart, but the Kotlin API names no WIT: records become data classes, variants sealed interfaces, resources AutoCloseable interfaces, and a call that can fail returns HcertResult<T, E>. Each exported interface becomes a Kotlin interface the library implements, for example Codec or Verifier; imported ones (Signer, Terminology) are implemented by the host application. The entry point is object Hcert: Hcert.verifier, Hcert.issuer and so on.

The module targets the JVM (Java 11+), which Android uses as-is, and Node.js through Kotlin/JS IR with generated TypeScript declarations. Regenerate with python3 tools/wit2kotlin/wit2kotlin.py; CI fails if the API is out of date with the contract.

HcertResult.kt

source
// GENERATED by tools/wit2kotlin/wit2kotlin.py. Do not edit.
package who.smart.trust

/**
 * The outcome of a call that can fail in an expected way: the value, or an
 * error saying which step failed and why.
 *
 * Expected failures are values, not exceptions [STT-VER-008], so the JVM and
 * Android API returns this instead of throwing. Kotlin's own `Result` cannot
 * carry a typed error, hence this type. The Kotlin/JS facade may throw, to keep
 * one TypeScript surface [STT-GEN-011].
 */
sealed interface HcertResult<out T, out E> {
    data class Ok<out T>(val value: T) : HcertResult<T, Nothing>

    data class Err<out E>(val error: E) : HcertResult<Nothing, E>
}

inline fun <T, E, R> HcertResult<T, E>.fold(onOk: (T) -> R, onErr: (E) -> R): R = when (this) {
    is HcertResult.Ok -> onOk(value)
    is HcertResult.Err -> onErr(error)
}

fun <T, E> HcertResult<T, E>.getOrNull(): T? = (this as? HcertResult.Ok)?.value

sealed interface CodecError {

     * Character outside the alphabet, dangling character or overflow.
     * [STT-B45-002]
     */
    data class Base45Invalid(val value: String) : CodecError

    data class InflateFailed(val value: String) : CodecError

    /** A decode-limits bound was hit. [STT-CBR-003, STT-ZIP-002] */
    data class LimitExceeded(val value: String) : CodecError

    data class CborMalformed(val value: String) : CodecError

    /** No `HC[1-9A-Z]:` context identifier. [STT-HC1-001] */
    data object BadPrefix : CodecError

    /** A well-formed context identifier other than HC1, e.g. "HC2". */
    data class UnsupportedContextVersion(val value: String) : CodecError
}

/**
 * Steps 4, 6, 7, 8: CBOR, ZLIB, Base45 and the HC1 context identifier.
 * Pure functions: no crypto, no I/O, no clock.
 *
 * Implemented by the library.
 */
interface Codec {
    /** RFC 9285 Base45. [STT-B45-001] */
    fun base45Encode(data: ByteArray): String

    fun base45Decode(text: String): HcertResult<ByteArray, CodecError>

    /** ZLIB (RFC 1950) wrapping DEFLATE (RFC 1951). [STT-ZIP-001] */
    fun zlibCompress(data: ByteArray, level: Int): ByteArray

    fun zlibInflate(data: ByteArray, maxOut: Int): HcertResult<ByteArray, CodecError>

    /**
     * Returns the context identifier version character (e.g. "1") and
     * the remainder after the colon. [STT-HC1-001]
     */
    fun detectContext(text: String): HcertResult<Pair<String, String>, CodecError>

    /**
     * Steps 6 to 8: COSE bytes to "HC1:..." string. The result uses only
     * QR Alphanumeric characters. [STT-QR-001]
     */
    fun toHc1(cose: ByteArray): String

    /** Steps 8 to 6 reversed: "HC1:..." string to COSE bytes. */
    fun fromHc1(text: String, limits: DecodeLimits): HcertResult<ByteArray, CodecError>

    /**
     * Re-encode CBOR using RFC 8949 core deterministic encoding.
     * [STT-CBR-001]
     */
    fun cborDeterministic(cbor: CborBytes, limits: DecodeLimits): HcertResult<CborBytes, CodecError>
}

    data class DevelopmentSubclaimNotAllowed(val value: Int) : IssueError

    data class UnknownSubclaim(val value: Int) : IssueError

    data object ValidityOutsideDsc : IssueError

    data class BadSignatureFormat(val value: String) : IssueError

    /** Raw COSE_Sign1 for 8-bit-safe transports. [STT-GEN-009] */
    val cose: ByteArray,
)

interface PendingSignature : AutoCloseable {
    /** Bytes to sign (CBOR Sig_structure). [STT-COS-004] */
    fun toBeSigned(): ByteArray

    fun algorithm(): CoseAlgorithm

    fun kid(): Kid
}

/**
 * End-to-end issuance (steps 3 to 8).
 *
 * Signing is two-phase so that any key holder can sign without the core
 * calling back into the host: prepare() returns the Sig_structure bytes,
 * the caller signs them with an HSM, KMS, Android Keystore or WebCrypto,
 * and finish() assembles the certificate. [STT-ISS-004, STT-COS-006]
 *
 * Implemented by the library.
 */
interface Issuer {
    /** Phase 1: validate, build CWT, CBOR and protected header. */
    fun prepare(req: IssueRequest, reg: Registry): HcertResult<PendingSignature, IssueError>

    /**
     * Phase 2: attach the signature. ES256 accepts r||s or DER and always
     * emits r||s. [STT-COS-005]
     */
    fun finish(pending: PendingSignature, signature: ByteArray): HcertResult<IssuedHcert, IssueError>
}

     */
    val url: String? = null,
    /** List.entry.item.reference, when it was not; the host reads it first. */
    val reference: String? = null,
    /** The decrypted content type (attachment.contentType). */
    val contentType: String? = null,
    val title: String? = null,
)

data class Decrypted(
    /** The JWE cty header, when present. */
    val contentType: String? = null,
    val content: ByteArray,
)

/** What the receiver signs for the HTTP Message Signatures option. */
data class SignatureRequest(
    val method: String,
    val authority: String,
    val path: String,
    val contentType: String? = null,
    val body: ByteArray? = null,
    val keyid: String,
    /** e.g. ecdsa-p256-sha256 */
    val alg: String,
    val created: NumericDate,
)

data class SignatureBase(
    /** RFC 9530 Content-Digest header value, when there is a body. */
    val contentDigest: String? = null,
    /** Signature-Input header value. */
    val signatureInput: String,
    /** The RFC 9421 signature base: the bytes the host signs. */
    val base: ByteArray,
)

/**
 * Verifiable Health Links, as the IHE ITI VHL profile specifies them, with
 * the SMART Health Links payload format. Pure functions: the host does the
 * HTTP (ITI-YY5 manifest search, ITI-68 document GETs) over mutual TLS, holds
 * the receiver's signing key and logs audit events. [STT-VHL-010]
 *
 * Implemented by the library.
 */
interface Vhl {
    /**
     * Every rule of the payload format; with `now`, an expired link too.
     * [STT-VHL-001, STT-VHL-009]
     */
    fun validate(p: VhlPayload, now: NumericDate?): List<ValidationIssue>

    /** [STT-VHL-003] */
    fun toVhlink(p: VhlPayload): String

    fun fromVhlink(text: String): HcertResult<VhlPayload, VhlError>

    /** A new key from the platform's secure random generator. [STT-VHL-004] */
    fun newKey(): String

    /** [STT-VHL-005] */
    fun buildManifestRequest(p: VhlPayload, recipient: String, passcode: String?, embeddedLengthMax: Int?): HcertResult<ManifestRequest, VhlError>

    /** The ITI-YY5 searchset Bundle. [STT-VHL-006] */
    fun parseManifest(bundleJson: String): HcertResult<List<ManifestDocument>, VhlError>

    /** JWE Compact, dir / A256GCM, the payload key. [STT-VHL-007] */
    fun encrypt(content: ByteArray, contentType: String, key: String): HcertResult<String, VhlError>

    fun decrypt(jwe: String, key: String): HcertResult<Decrypted, VhlError>

    /** [STT-VHL-008] */
    fun buildSignatureBase(r: SignatureRequest): SignatureBase

    /**
     * The Signature header value. An ecdsa-p256-sha256 DER signature is
     * converted to r||s.
     */
    fun signatureHeader(signature: ByteArray, alg: String): HcertResult<String, VhlError>
}
