Kotlin API skeleton

Declarations only, with no behaviour. tools/wit2kotlin.py generates them from the WIT contract using the mapping rules: records become data classes, variants become sealed interfaces, resources become AutoCloseable interfaces, and result<T, E> becomes WitResult<T, E>. Each exported WIT interface becomes a Kotlin interface that the library implements, for example Codec or Verifier. Imported interfaces (Signer, Terminology) are implemented by the host application. Each WIT world becomes a bundle interface, plus a factory that takes the world's imports.

The module targets the JVM (Java 11+), which Android uses as-is, and Node.js through Kotlin/JS IR with generated TypeScript declarations. Regenerate with python3 tools/wit2kotlin.py; CI fails if the skeleton is out of date with the WIT.

// GENERATED by tools/wit2kotlin.py. Do not edit.
package who.smart.trust

/**
 * Kotlin form of WIT `result<T, E>`.
 *
 * Expected failures are values, not exceptions [STT-VER-008], so the JVM and
 * Android API returns this instead of throwing. The Kotlin/JS facade may throw
 * like jco does, to keep one TypeScript surface [STT-GEN-011].
 */
sealed interface WitResult<out T, out E> {
    data class Ok<out T>(val value: T) : WitResult<T, Nothing>

    data class Err<out E>(val error: E) : WitResult<Nothing, E>
}

inline fun <T, E, R> WitResult<T, E>.fold(onOk: (T) -> R, onErr: (E) -> R): R = when (this) {
    is WitResult.Ok -> onOk(value)
    is WitResult.Err -> onErr(error)
}

fun <T, E> WitResult<T, E>.getOrNull(): T? = (this as? WitResult.Ok)?.value

package who.smart.trust.codec

import who.smart.trust.WitResult
import who.smart.trust.types.CborBytes
import who.smart.trust.types.DecodeLimits

sealed interface CodecError {
    /**
     * Character outside the alphabet, dangling character or overflow.
     * [STT-B45-002]
     */
    data class Base45Invalid(val value: String) : CodecError

    data class InflateFailed(val value: String) : CodecError

    /** A decode-limits bound was hit. [STT-CBR-003, STT-ZIP-002] */
    data class LimitExceeded(val value: String) : CodecError

    data class CborMalformed(val value: String) : CodecError

    /** No `HC[1-9A-Z]:` context identifier. [STT-HC1-001] */
    data object BadPrefix : CodecError

    /** A well-formed context identifier other than HC1, e.g. "HC2". */
    data class UnsupportedContextVersion(val value: String) : CodecError
}

/**
 * Steps 4, 6, 7, 8: CBOR, ZLIB, Base45 and the HC1 context identifier.
 * Pure functions: no crypto, no I/O, no clock.
 *
 * Implemented by the library (Kotlin and Rust) for the `codec` WIT interface.
 */
interface Codec {
    /** RFC 9285 Base45. [STT-B45-001] */
    fun base45Encode(data: ByteArray): String

    fun base45Decode(text: String): WitResult<ByteArray, CodecError>

    /** ZLIB (RFC 1950) wrapping DEFLATE (RFC 1951). [STT-ZIP-001] */
    fun zlibCompress(data: ByteArray, level: Int): ByteArray

    fun zlibInflate(data: ByteArray, maxOut: Int): WitResult<ByteArray, CodecError>

    /**
     * Returns the context identifier version character (e.g. "1") and
     * the remainder after the colon. [STT-HC1-001]
     */
    fun detectContext(text: String): WitResult<Pair<String, String>, CodecError>

    /** Steps 6 to 8: COSE bytes to "HC1:..." string. */
    fun toHc1(cose: ByteArray): String

    /** Steps 8 to 6 reversed: "HC1:..." string to COSE bytes. */
    fun fromHc1(text: String, limits: DecodeLimits): WitResult<ByteArray, CodecError>

    /**
     * Re-encode CBOR using RFC 8949 core deterministic encoding.
     * [STT-CBR-001]
     */
    fun cborDeterministic(cbor: CborBytes, limits: DecodeLimits): WitResult<CborBytes, CodecError>

    /**
     * RFC 8949 diagnostic notation, for display and debugging only.
     * [STT-GEN-008]
     */
    fun cborDiagnostic(cbor: CborBytes, limits: DecodeLimits): WitResult<String, CodecError>
}

    val hc1: String,
    /** Raw COSE_Sign1 for 8-bit-safe transports. [STT-GEN-009] */
    val cose: ByteArray,
    val qr: QrMatrix? = null,
    val report: EncodingReport,
)

interface PendingSignature : AutoCloseable {
    /** Bytes to sign (CBOR Sig_structure). [STT-COS-004] */
    fun toBeSigned(): ByteArray

    fun algorithm(): CoseAlgorithm

    fun kid(): Kid
}

/**
 * End-to-end issuance (steps 3 to 9).
 *
 * Signing is two-phase so that any key holder can sign without the core
 * calling back into the host: prepare() returns the Sig_structure bytes,
 * the caller signs them with an HSM, KMS, Android Keystore or WebCrypto,
 * and finish() assembles the certificate. [STT-ISS-004, STT-COS-006]
 *
 * Implemented by the library (Kotlin and Rust) for the `issuer` WIT interface.
 */
interface Issuer {
    /** Phase 1: validate, build CWT, CBOR and protected header. */
    fun prepare(req: IssueRequest, reg: Registry, policy: IssuerPolicy): WitResult<PendingSignature, IssueError>

    /**
     * Phase 2: attach the signature. ES256 accepts r||s or DER and always
     * emits r||s. [STT-COS-005]
     */
    fun finish(pending: PendingSignature, signature: ByteArray): WitResult<IssuedHcert, IssueError>

    /** One call, using the host `signer` import. [STT-ISS-001] */
    fun issue(req: IssueRequest, reg: Registry, policy: IssuerPolicy): WitResult<IssuedHcert, IssueError>
}
